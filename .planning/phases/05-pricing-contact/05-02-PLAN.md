```javascript
// Fetch and render content from data/content.json
document.addEventListener('DOMContentLoaded', function() {
  fetch('data/content.json')
    .then(response => {
      if (!response.ok) {
        throw new Error('Failed to load content.json: ' + response.status);
      }
      return response.json();
    })
    .then(data => {
      renderPricing(data.pricing);
      renderContact(data.contact);
    })
    .catch(error => {
      console.error('Error loading content:', error);
      // Fallback: show error message but don't break page
      const pricingGrid = document.getElementById('pricing-grid');
      if (pricingGrid) {
        pricingGrid.textContent = 'Unable to load pricing data. Please try refreshing the page.';
      }
    });
});

// Render pricing tiers into #pricing-grid
function renderPricing(pricingData) {
  const pricingIntro = document.getElementById('pricing-intro');
  const pricingGrid = document.getElementById('pricing-grid');
  
  if (!pricingGrid) return;
  
  // Render intro text (text node, XSS-safe)
  if (pricingIntro && pricingData.intro) {
    pricingIntro.textContent = pricingData.intro;
  }
  
  // Clear any placeholder content
  pricingGrid.innerHTML = '';
  
  // Render each pricing tier as a card
  pricingData.tiers.forEach((tier, index) => {
    const card = document.createElement('div');
    card.className = 'pricing-card';
    
    // Add 'featured' class to elite tier (optional for styling)
    if (tier.id === 'elite') {
      card.classList.add('featured');
    }
    
    // Build card content using text nodes (XSS-safe)
    
    // Tier name (h3)
    const nameEl = document.createElement('h3');
    nameEl.textContent = tier.name;
    card.appendChild(nameEl);
    
    // Price display
    const priceEl = document.createElement('div');
    priceEl.className = 'price';
    
    const priceNumber = document.createElement('span');
    priceNumber.className = 'price-number';
    priceNumber.textContent = '₹' + tier.price.toLocaleString('en-IN');
    priceEl.appendChild(priceNumber);
    
    const durationEl = document.createElement('span');
    durationEl.className = 'price-duration';
    durationEl.textContent = tier.duration;
    priceEl.appendChild(durationEl);
    
    card.appendChild(priceEl);
    
    // Discount label (if present)
    if (tier.discount_label) {
      const discountEl = document.createElement('p');
      discountEl.className = 'discount-label';
      discountEl.textContent = tier.discount_label;
      card.appendChild(discountEl);
    }
    
    // Feature list (ul > li with text nodes)
    const featureList = document.createElement('ul');
    featureList.className = 'features';
    
    tier.features.forEach(feature => {
      const featureItem = document.createElement('li');
      featureItem.textContent = feature;
      featureList.appendChild(featureItem);
    });
    
    card.appendChild(featureList);
    
    // CTA Button
    const btnContainer = document.createElement('div');
    btnContainer.className = 'cta-container';
    
    const ctaBtn = document.createElement('button');
    ctaBtn.className = 'cta-button';
    ctaBtn.textContent = 'Select Plan';
    ctaBtn.addEventListener('click', function() {
      // Scroll to contact section (link to #contact for now)
      document.getElementById('contact').scrollIntoView({ behavior: 'smooth' });
    });
    btnContainer.appendChild(ctaBtn);
    card.appendChild(btnContainer);
    
    pricingGrid.appendChild(card);
  });
}
```

```javascript
// Expand renderContact() function (replace placeholder)
function renderContact(contactData) {
  const contactInfo = document.getElementById('contact-info');
  if (!contactInfo) return;
  
  // Clear placeholder
  contactInfo.innerHTML = '';
  
  // Address
  const addressEl = document.createElement('div');
  addressEl.className = 'contact-item';
  
  const addressLabel = document.createElement('h4');
  addressLabel.textContent = 'Address';
  addressEl.appendChild(addressLabel);
  
  const addressText = document.createElement('p');
  addressText.textContent = contactData.address;
  addressEl.appendChild(addressText);
  contactInfo.appendChild(addressEl);
  
  // Phone
  const phoneEl = document.createElement('div');
  phoneEl.className = 'contact-item';
  
  const phoneLabel = document.createElement('h4');
  phoneLabel.textContent = 'Phone';
  phoneEl.appendChild(phoneLabel);
  
  const phoneLink = document.createElement('a');
  phoneLink.href = 'tel:' + contactData.phone;
  phoneLink.textContent = contactData.phone;
  phoneEl.appendChild(phoneLink);
  contactInfo.appendChild(phoneEl);
  
  // Email
  const emailEl = document.createElement('div');
  emailEl.className = 'contact-item';
  
  const emailLabel = document.createElement('h4');
  emailLabel.textContent = 'Email';
  emailEl.appendChild(emailLabel);
  
  const emailLink = document.createElement('a');
  emailLink.href = 'mailto:' + contactData.email;
  emailLink.textContent = contactData.email;
  emailEl.appendChild(emailLink);
  contactInfo.appendChild(emailEl);
  
  // Hours
  const hoursEl = document.createElement('div');
  hoursEl.className = 'contact-item';
  
  const hoursLabel = document.createElement('h4');
  hoursLabel.textContent = 'Hours';
  hoursEl.appendChild(hoursLabel);
  
  const weekdaysText = document.createElement('p');
  weekdaysText.textContent = 'Weekdays: ' + contactData.hours.weekdays;
  hoursEl.appendChild(weekdaysText);
  
  const weekendText = document.createElement('p');
  weekendText.textContent = 'Weekends: ' + contactData.hours.weekends;
  hoursEl.appendChild(weekendText);
  
  contactInfo.appendChild(hoursEl);
  
  // Set up form handler
  setupContactForm();
}

// Form submission handler
function setupContactForm() {
  const contactForm = document.getElementById('contact-form');
  const formMessage = document.getElementById('form-message');
  
  if (!contactForm) return;
  
  contactForm.addEventListener('submit', function(e) {
    e.preventDefault();
    
    // Client-side validation (HTML5 required already handles this, but explicit check)
    const name = contactForm.querySelector('input[name="name"]').value.trim();
    const email = contactForm.querySelector('input[name="email"]').value.trim();
    const message = contactForm.querySelector('textarea[name="message"]').value.trim();
    
    if (!name || !email || !message) {
      if (formMessage) {
        formMessage.textContent = 'Please fill in all fields.';
        formMessage.className = 'form-message error';
      }
      return;
    }
    
    // Show success message (client-side only — no backend integration yet, per D-05)
    if (formMessage) {
      formMessage.textContent = 'Thank you! We\'ll get back to you soon.';
      formMessage.className = 'form-message success';
    }
    
    // Log form data (for debugging; can be removed later)
    console.log('Form submitted:', { name, email, message });
    
    // Clear form fields
    contactForm.reset();
    
    // Clear message after 3 seconds
    setTimeout(function() {
      if (formMessage) {
        formMessage.textContent = '';
        formMessage.className = 'form-message';
      }
    }, 3000);
  });
}
```
